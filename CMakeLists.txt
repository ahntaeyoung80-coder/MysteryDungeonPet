cmake_minimum_required(VERSION 3.20)
project(MysteryDungeonPet VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(pet_core src/core/package.cpp src/core/animator.cpp src/core/surface.cpp
  src/core/exploration.cpp src/core/gameplay.cpp src/core/gameplay_scene.cpp src/core/guild_scene.cpp src/core/state_store.cpp src/core/exploration_package.cpp)
target_include_directories(pet_core PUBLIC src)
include(CTest)
if(BUILD_TESTING)
  add_executable(pet_revision_tests tests/revision_tests.cpp)
  target_link_libraries(pet_revision_tests PRIVATE pet_core)
  add_test(NAME pet_revision_tests COMMAND pet_revision_tests)
  add_executable(pet_improvement_tests tests/improvement_tests.cpp)
  target_link_libraries(pet_improvement_tests PRIVATE pet_core)
  add_test(NAME pet_improvement_tests COMMAND pet_improvement_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets/original/gameplay/actors/asset-0/asset.mdpa")
  add_executable(pet_gameplay_tests tests/gameplay_tests.cpp)
  target_link_libraries(pet_gameplay_tests PRIVATE pet_core)
  add_test(NAME pet_gameplay_tests COMMAND pet_gameplay_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets/original/gameplay")
  add_executable(pet_exploration_tests tests/exploration_tests.cpp)
  target_link_libraries(pet_exploration_tests PRIVATE pet_core)
  add_executable(pet_explore tools/explore.cpp)
  target_link_libraries(pet_explore PRIVATE pet_core)
  add_test(NAME pet_exploration_tests COMMAND pet_exploration_tests
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/original/exploration/dungeon-0/exploration.mdpe"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/original/exploration/dungeon-15/exploration.mdpe")
  add_executable(pet_package_inspect tests/package_inspect.cpp)
  target_link_libraries(pet_package_inspect PRIVATE pet_core)
  add_executable(pet_core_tests tests/core_tests.cpp)
  target_link_libraries(pet_core_tests PRIVATE pet_core)
  add_executable(pet_package_playback tests/package_playback.cpp)
  target_link_libraries(pet_package_playback PRIVATE pet_core)
  add_test(NAME pet_core_tests COMMAND pet_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets/fixture/fixture.mdpa")
  add_test(NAME pet_package_playback COMMAND pet_package_playback "${CMAKE_CURRENT_SOURCE_DIR}/assets/fixture/fixture.mdpa")
endif()
if(WIN32)
  enable_language(RC)
  add_executable(MysteryDungeonPet WIN32 src/windows/main.cpp src/windows/assets.cpp src/windows/app.rc)
  target_include_directories(MysteryDungeonPet PRIVATE src/windows)
  target_compile_definitions(MysteryDungeonPet PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(MysteryDungeonPet PRIVATE pet_core user32 gdi32 ole32 windowscodecs wtsapi32 shell32 uuid psapi bcrypt)
  if(MSVC)
    # app.rc embeds our PMv2 manifest; disable the linker's generated manifest.
    target_link_options(MysteryDungeonPet PRIVATE /MANIFEST:NO)
  endif()
  if(MINGW)
    target_link_options(MysteryDungeonPet PRIVATE -municode -static -static-libgcc -static-libstdc++ $<$<CONFIG:Release>:-s>)
  endif()
  add_custom_command(TARGET MysteryDungeonPet POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/fixture" "$<TARGET_FILE_DIR:MysteryDungeonPet>/assets/fixture"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:MysteryDungeonPet>/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/third_party/Windows-classic-samples-LICENSE.txt" "$<TARGET_FILE_DIR:MysteryDungeonPet>/licenses/Windows-classic-samples-LICENSE.txt")
  add_custom_command(TARGET MysteryDungeonPet POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/original/runtime" "$<TARGET_FILE_DIR:MysteryDungeonPet>/assets/original")
  add_custom_command(TARGET MysteryDungeonPet POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/original/exploration" "$<TARGET_FILE_DIR:MysteryDungeonPet>/assets/exploration")
  add_custom_command(TARGET MysteryDungeonPet POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/original/gameplay" "$<TARGET_FILE_DIR:MysteryDungeonPet>/assets/gameplay")
  install(TARGETS MysteryDungeonPet RUNTIME DESTINATION .)
  install(DIRECTORY assets/fixture DESTINATION assets)
  install(DIRECTORY assets/original/runtime/ DESTINATION assets/original)
  install(DIRECTORY assets/original/exploration/ DESTINATION assets/exploration)
  install(DIRECTORY assets/original/gameplay/ DESTINATION assets/gameplay)
  install(FILES third_party/Windows-classic-samples-LICENSE.txt DESTINATION licenses)
  install(FILES tools/Run-A2.ps1 DESTINATION .)
  install(FILES tools/Run-B1.ps1 docs/B1_USAGE.md DESTINATION .)
  install(FILES docs/WINDOWS_QUICKSTART.md DESTINATION .)
  install(FILES docs/A3_USAGE.md docs/A3_WINDOWS_QUICKSTART.md DESTINATION .)
endif()
