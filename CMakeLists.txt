  enable_language(RC)

  target_compile_definitions(pet_text_overlay PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(pet_text_overlay PUBLIC gdi32)
  if(BUILD_TESTING)
    add_executable(pet_text_overlay_tests tests/text_overlay_windows_tests.cpp)
    target_link_libraries(pet_text_overlay_tests PRIVATE pet_text_overlay pet_core)
    add_test(NAME pet_text_overlay_tests COMMAND pet_text_overlay_tests)
  endif()
  add_executable(MysteryDungeonPet WIN32 src/windows/main.cpp src/windows/assets.cpp src/windows/app.rc)
  target_include_directories(MysteryDungeonPet PRIVATE src/windows)
  target_compile_definitions(MysteryDungeonPet PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(MysteryDungeonPet PRIVATE pet_core pet_text_overlay user32 gdi32 ole32 windowscodecs wtsapi32 shell32 uuid psapi bcrypt)
  if(MSVC)
    # app.rc embeds our PMv2 manifest; disable the linker's generated manifest.
